src/

│
├── client/
│   ├── ClientMain.java             // Main per avviare il Nodo Sensore

│   ├── PeerServer.java             // ServerSocket interno al nodo per servire gli altri peer
│   ├── PeerClientHandler.java      // Thread gestore dell'invio risorsa a un altro peer
│   ├── LocalStorage.java           // Gestore delle rilevazioni locali (thread-safe)
│   └── RobustDownloader.java       // Implementazione del protocollo di download tollerante ai guasti
│
└── common/
    └── Protocol.java               // Messaggi di rete, costanti, comandi e classi di supporto (es. DTO)
